f = open('27A (6).txt')
a = [list(map(float, f.readline().split())) for i in range(660)]
c1 = []
c2 = []
c3 = []
for x, y in a:
    if 10<=x<=20 and 0<=y<=10:
        c1.append([x,y])
    elif 0<=x<=9 and 20<=y<=30:
        c2.append([x, y])
    elif 20<=x<=28 and 15<=y<=25:
        c3.append([x,y])
def d(cen, pr):
    return ((pr[0] - cen[0])**2 + (pr[1] - cen[1])**2)**0.5

centers = []
min = 10**10
last = 0
for cen in c1:
    sum = 0
    for pr in c1:
        sum+=d(cen,pr)
    if sum < min:
        min = sum
        last = cen
centers.append(last)
min = 10**10
last = 0
for cen in c2:
    sum = 0
    for pr in c2:
        sum+=d(cen,pr)
    if sum < min:
        min = sum
        last = cen
centers.append(last)
min = 10**10
last = 0
for cen in c3:
    sum = 0
    for pr in c3:
        sum+=d(cen,pr)
    if sum < min:
        min = sum
        last = cen
centers.append(last)
print(int((centers[0][0]+centers[1][0]+centers[2][0])/3*10_000), int((centers[0][1]+centers[1][1]+centers[2][1])/3*10_000))
